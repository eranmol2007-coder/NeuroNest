━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  🚀 TO START YOUR WEBSITE (ONE CLICK):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Double-click this file:
    
    ► 🚀 START NEURONEST.bat
    
    That's it! The website will open automatically in your
    browser at: http://localhost:5175


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ℹ️  WHAT HAPPENS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    1. Backend server starts (minimized window)
    2. Frontend server starts (minimized window)
    3. Waits 10 seconds for servers to be ready
    4. Browser opens automatically
    5. NeuroNest loads!


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  🛑 TO STOP THE SERVERS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    You'll see two minimized windows on your taskbar:
    
    • "NeuroNest Backend"
    • "NeuroNest Frontend"
    
    Just close both windows to stop the servers.
    
    OR
    
    Open Task Manager (Ctrl+Alt+Delete) and end the
    Node.js processes.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ⚠️  FIRST TIME SETUP:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    If this is your FIRST TIME, install dependencies:
    
    1. Open Command Prompt in this folder
    2. Run: cd backend && npm install
    3. Run: cd ../frontend && npm install
    4. Then double-click: 🚀 START NEURONEST.bat


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  🌐 URLS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ❓ TROUBLESHOOTING:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    ❌ Page doesn't load?
       → Wait 15 seconds, then refresh browser
    
    ❌ Port already in use?
       → Close existing Node.js processes
       → Try again
    
    ❌ Errors in terminal?
       → Make sure you ran: npm install
       → In both backend AND frontend folders
